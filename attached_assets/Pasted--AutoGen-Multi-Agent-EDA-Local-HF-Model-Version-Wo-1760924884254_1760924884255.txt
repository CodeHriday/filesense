import torch, gc
import pandas as pd
import seaborn as sns
import matplotlib.pyplot as plt
from sklearn.datasets import load_iris

# Free memory utility (for Colab stability)
def clear_memory():
    gc.collect()
    if torch.cuda.is_available():
        torch.cuda.empty_cache()

device = "cuda" if torch.cuda.is_available() else "cpu"
print(f"Running on device: {device}")

# ==========================================
# 2. Load a Small Dataset
# ==========================================
iris = load_iris(as_frame=True)
df = iris.frame
df.to_csv("dataset.csv", index=False)
print("Sample dataset saved as dataset.csv")

# ==========================================
# 3. Custom Lightweight Model Client
# ==========================================
from langchain_huggingface import ChatHuggingFace, HuggingFacePipeline
from autogen_agentchat.agents import AssistantAgent, UserProxyAgent, CodeExecutorAgent
from autogen_agentchat.teams import RoundRobinGroupChat
from autogen_agentchat.conditions import TextMentionTermination
from autogen_ext.code_executors.local import LocalCommandLineCodeExecutor

# Use a small, memory-friendly model
# You may replace it with a slightly larger model if GPU is available.
MODEL_ID = "mistralai/Mistral-7B-Instruct-v0.1" if torch.cuda.is_available() else "HuggingFaceH4/zephyr-7b-alpha"

hf_pipeline = HuggingFacePipeline.from_model_id(
    model_id=MODEL_ID,
    task="text-generation",
    device=device,
    pipeline_kwargs={
        "max_new_tokens": 256,
        "temperature": 0.3,
        "do_sample": False
    }
)
local_model = ChatHuggingFace(llm=hf_pipeline)
clear_memory()

# ==========================================
# 4. Define Multi-Agent Workflow
# ==========================================
data_preparer = AssistantAgent(
    "data_preparer",
    model_client=local_model,
    system_message=(
        "You prepare datasets by handling missing values, standardizing column names, "
        "and printing brief summaries. Keep memory usage low."
    ),
)

eda_analyst = AssistantAgent(
    "eda_analyst",
    model_client=local_model,
    system_message=(
        "Perform lightweight EDA — basic statistics, correlations, and small sample visualizations. "
        "Avoid large plot generation to conserve memory."
    ),
)

report_generator = AssistantAgent(
    "report_generator",
    model_client=local_model,
    system_message=(
        "Summarize findings from EDA into a markdown-style final report, highlighting insights and recommendations."
    ),
)

critic = AssistantAgent(
    "critic",
    model_client=local_model,
    system_message=(
        "Review outputs for readability, correctness, and clarity. Flag issues and suggest fixes."
    ),
)

executor = CodeExecutorAgent(
    "executor",
    code_executor=LocalCommandLineCodeExecutor()
)

admin = UserProxyAgent("admin")
termination = TextMentionTermination("end_eda", sources=["admin"])

eda_team = RoundRobinGroupChat(
    [admin, data_preparer, eda_analyst, report_generator, critic, executor],
    termination_condition=termination,
)

# ==========================================
# 5. Simulated Execution Loop
# ==========================================
# Because running all-agent conversations with local models can be memory intensive,
# here we simulate the cooperative flow in short steps.

print("\n=== PHASE 1: Data Preparation ===")
data_summary = """
The dataset has the following columns: sepal length, sepal width, petal length, petal width, and species.
There are no missing values.
"""
print(data_summary)
clear_memory()

print("\n=== PHASE 2: EDA Analysis ===")
sns.pairplot(df, hue="target")
plt.show()
clear_memory()

print("\n=== PHASE 3: Report Generation ===")
eda_report = """
# EDA Summary Report

**Dataset Overview**
- 150 samples across 5 columns.
- No missing values detected.

**Key Insights**
- Petal dimensions are most strongly correlated with flower species.
- Setosa species exhibits smallest petal length and width.

**Visual Insights**
Scatterplots and pairwise comparisons show clear separability between species groups.
"""
print(eda_report)
clear_memory()

# ==========================================
# 6. Save Final EDA Report
# ==========================================
with open("eda_final_report.md", "w") as f:
    f.write(eda_report)
print("EDA final report saved as eda_final_report.md")

# ==========================================
# 7. Optional: HTML Report Generation
# ==========================================
from ydata_profiling import ProfileReport
profile = ProfileReport(df, title="Compact EDA Report", minimal=True)
profile.to_file("eda_profile_report.html")
print("Lightweight HTML report saved as eda_profile_report.html")

clear_memory()
