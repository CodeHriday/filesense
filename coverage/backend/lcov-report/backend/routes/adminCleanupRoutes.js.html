/**
 * Admin Cleanup Routes
 * 
 * Provides endpoints for admin users to perform system-wide data cleanup.
 * Preserves critical system tables (users, roles, permissions) while allowing
 * deletion of operational data in categories.
 */
 
const express = require('express');
const router = express.Router();
 
/**
 * Get record counts for all cleanup categories
 * Helps admin preview what will be deleted before confirmation
 */
router.get('/counts', async (req, res) => {
  const pool = req.app.locals.pool;
  
  try {
    const counts = {};
    
    // Files & Storage category
    const filesCount = await pool.query('SELECT COUNT(*) FROM files');
    const fileColumnsCount = await pool.query('SELECT COUNT(*) FROM file_columns');

      files: parseInt(filesCount.rows[0].count),
      file_columns: parseInt(fileColumnsCount.rows[0].count),
      file_preview_data: parseInt(filePreviewCount.rows[0].count),
      file_schemas: parseInt(fileSchemasCount.rows[0].count)
    };
    
    // Transformations & ETL category
    const transformationsCount = await pool.query('SELECT COUNT(*) FROM transformation_mappings');
    const jobsCount = await pool.query('SELECT COUNT(*) FROM transformation_jobs');
    const logsCount = await pool.query('SELECT COUNT(*) FROM transformation_logs');
    const profilingCount = await pool.query('SELECT COUNT(*) FROM profiling_metrics');
    const anomaliesCount = await pool.query('SELECT COUNT(*) FROM anomalies');
    
    counts.transformations = {
      transformation_mappings: parseInt(transformationsCount.rows[0].count),
      transformation_jobs: parseInt(jobsCount.rows[0].count),
      transformation_logs: parseInt(logsCount.rows[0].count),
      profiling_metrics: parseInt(profilingCount.rows[0].count),
      anomalies: parseInt(anomaliesCount.rows[0].count)
    };
    
    // Workflows & Orchestration category
    const workflowsCount = await pool.query('SELECT COUNT(*) FROM n_step_workflows');
    const workflowRunsCount = await pool.query('SELECT COUNT(*) FROM n_step_workflow_runs');
    const workflowStepsNStepCount = await pool.query('SELECT COUNT(*) FROM n_step_workflow_steps');
    const workflowStepsCount = await pool.query('SELECT COUNT(*) FROM workflow_steps');
    const mappingGroupsCount = await pool.query('SELECT COUNT(*) FROM mapping_groups');
    const mappingGroupExecutionsCount = await pool.query('SELECT COUNT(*) FROM mapping_group_executions');
    const mappingGroupMembersCount = await pool.query('SELECT COUNT(*) FROM mapping_group_members');
    const schedulesCount = await pool.query('SELECT COUNT(*) FROM schedules');
    
    counts.workflows = {
      n_step_workflows: parseInt(workflowsCount.rows[0].count),
      n_step_workflow_runs: parseInt(workflowRunsCount.rows[0].count),
      n_step_workflow_steps: parseInt(workflowStepsNStepCount.rows[0].count),
      workflow_steps: parseInt(workflowStepsCount.rows[0].count),
      mapping_groups: parseInt(mappingGroupsCount.rows[0].count),
      mapping_group_executions: parseInt(mappingGroupExecutionsCount.rows[0].count),
      mapping_group_members: parseInt(mappingGroupMembersCount.rows[0].count),
      schedules: parseInt(schedulesCount.rows[0].count)
    };
    
    // Data Governance category
    const columnGovernanceCount = await pool.query('SELECT COUNT(*) FROM column_governance_tag_assignments');
    const columnPoliciesCount = await pool.query('SELECT COUNT(*) FROM column_policy_assignments');
    const fileOwnershipCount = await pool.query('SELECT COUNT(*) FROM file_ownership');
    const tagsCount = await pool.query('SELECT COUNT(*) FROM governance_tags');
    const policiesCount = await pool.query('SELECT COUNT(*) FROM governance_policies');
    
    counts.governance = {
      column_governance_tag_assignments: parseInt(columnGovernanceCount.rows[0].count),
      column_policy_assignments: parseInt(columnPoliciesCount.rows[0].count),
      file_ownership: parseInt(fileOwnershipCount.rows[0].count),
      governance_tags: parseInt(tagsCount.rows[0].count),
      governance_policies: parseInt(policiesCount.rows[0].count)
    };
    
    // Audit & Monitoring category
    const fileAuditCount = await pool.query('SELECT COUNT(*) FROM file_audit_details');
    const alertsCount = await pool.query('SELECT COUNT(*) FROM alert_history');
    const insightsCount = await pool.query('SELECT COUNT(*) FROM file_pattern_metrics');
    const executionAuditCount = await pool.query('SELECT COUNT(*) FROM execution_audit');
    const securityAuditCount = await pool.query('SELECT COUNT(*) FROM security_audit_log');
    
    counts.audits = {
      file_audit_details: parseInt(fileAuditCount.rows[0].count),
      alert_history: parseInt(alertsCount.rows[0].count),
      file_pattern_metrics: parseInt(insightsCount.rows[0].count),
      execution_audit: parseInt(executionAuditCount.rows[0].count),
      security_audit_log: parseInt(securityAuditCount.rows[0].count)
    };
    
    // Projects & Access Control category
    const projectsCount = await pool.query('SELECT COUNT(*) FROM projects WHERE id != 1'); // Exclude Default Project
    const projectMembersCount = await pool.query('SELECT COUNT(*) FROM project_members');
    
    counts.projects = {
      projects: parseInt(projectsCount.rows[0].count),
      project_members: parseInt(projectMembersCount.rows[0].count)
    };
    
    // Connections category

    // Groups category
    const groupsCount = await pool.query('SELECT COUNT(*) FROM access_groups WHERE id != 1'); // Exclude Default Group
    const groupMembersCount = await pool.query('SELECT COUNT(*) FROM access_group_members');
    const groupRolesCount = await pool.query('SELECT COUNT(*) FROM access_group_roles');
    
    counts.groups = {
      access_groups: parseInt(groupsCount.rows[0].count),
      access_group_members: parseInt(groupMembersCount.rows[0].count),

    const dbtExecutionsCount = await pool.query('SELECT COUNT(*) FROM dbt_executions');
    const dbtModelsCount = await pool.query('SELECT COUNT(*) FROM dbt_models');
    
    counts.dbt = {
      dbt_executions: parseInt(dbtExecutionsCount.rows[0].count),
      dbt_models: parseInt(dbtModelsCount.rows[0].count)
    };
    
    // Notifications & User Activity category
    const notificationsCount = await pool.query('SELECT COUNT(*) FROM notifications');
    const userSessionsCount = await pool.query('SELECT COUNT(*) FROM user_sessions');
    const userBadgesCount = await pool.query('SELECT COUNT(*) FROM user_badges');
    const userDashboardCount = await pool.query('SELECT COUNT(*) FROM user_dashboard_settings');
    
    counts.activity = {
      notifications: parseInt(notificationsCount.rows[0].count),
      user_sessions: parseInt(userSessionsCount.rows[0].count),
      user_badges: parseInt(userBadgesCount.rows[0].count),
      user_dashboard_settings: parseInt(userDashboardCount.rows[0].count)
    };
    
    // Quality Violations category
    const violationsCount = await pool.query('SELECT COUNT(*) FROM quality_rule_violations');
    
    counts.violations = {
      quality_rule_violations: parseInt(violationsCount.rows[0].count)
    };
    
    // Data Lineage category
    const lineageCount = await pool.query('SELECT COUNT(*) FROM transformation_lineage');
    
    counts.lineage = {
      transformation_lineage: parseInt(lineageCount.rows[0].count)
    };
    
    res.json({ counts });
  } catch (error) {
    console.error('Error fetching cleanup counts:', error);
    res.status(500).json({ 
      error: 'Failed to fetch cleanup counts',
      message: error.message 
    });
  }
});
 
/**
 * Execute system cleanup based on selected categories
 * Requires admin privileges and multiple confirmation
 */
router.post('/', async (req, res) => {
  const pool = req.app.locals.pool;
  const { categories, confirmationCode } = req.body;
  
  // Validate confirmation code
  if (confirmationCode !== 'DELETE ALL DATA') {
    return res.status(400).json({ 
      error: 'Invalid confirmation code',
      message: 'You must type "DELETE ALL DATA" to proceed'
    });
  }
  
  // Validate categories
  if (!Array.isArray(categories) || categories.length === 0) {
    return res.status(400).json({ 
      error: 'No categories selected',
      message: 'Please select at least one category to clean up'
    });
  }
  
  const validCategories = ['files', 'transformations', 'workflows', 'governance', 'audits', 'projects', 'connections', 'groups', 'vendors', 'dbt', 'activity', 'violations', 'lineage'];
  const invalidCategories = categories.filter(c => !validCategories.includes(c));
  
  if (invalidCategories.length > 0) {
    return res.status(400).json({ 
      error: 'Invalid categories',
      message: `Invalid categories: ${invalidCategories.join(', ')}`
    });
  }
  
  const client = await pool.connect();
  const deletedCounts = {};
  
  try {
    await client.query('BEGIN');
    
    // CROSS-CATEGORY DEPENDENCY CLEANUP
    // Delete dependent records from other categories to prevent FK violations
    
    // Before deleting files: clean up references from activity and vendors categories
    if (categories.includes('files')) {
      // Delete user_badges that reference files (even if activity category not selected)
      await client.query('DELETE FROM user_badges WHERE file_id IN (SELECT file_id FROM files)');
      // Delete vendor_file_profiling_results that reference files (even if vendors category not selected)
      await client.query('DELETE FROM vendor_file_profiling_results WHERE file_id IN (SELECT file_id FROM files)');
    }
    
    // Before deleting transformations: clean up file_audit_details that reference transformation_jobs
    if (categories.includes('transformations')) {
      // Delete file_audit_details that reference transformation_jobs (even if audits category not selected)
      await client.query('DELETE FROM file_audit_details WHERE transformation_job_id IN (SELECT id FROM transformation_jobs)');
    }
    
    // Before deleting connections: clean up vendor agreements and profiling that reference connections
    if (categories.includes('connections')) {
      // Delete vendor_file_profiling_results that reference storage connections
      await client.query('DELETE FROM vendor_file_profiling_results WHERE connection_id IN (SELECT connection_id FROM storage_connections)');
      // Delete vendor_agreements that reference storage connections (must delete children first)
      await client.query('DELETE FROM vendor_agreement_documents WHERE agreement_id IN (SELECT id FROM vendor_agreements WHERE vendor_connection_id IN (SELECT connection_id FROM storage_connections))');
      await client.query('DELETE FROM vendor_agreement_changes WHERE agreement_id IN (SELECT id FROM vendor_agreements WHERE vendor_connection_id IN (SELECT connection_id FROM storage_connections))');
      await client.query('DELETE FROM vendor_agreement_approvals WHERE agreement_id IN (SELECT id FROM vendor_agreements WHERE vendor_connection_id IN (SELECT connection_id FROM storage_connections))');
      await client.query('DELETE FROM vendor_agreements WHERE vendor_connection_id IN (SELECT connection_id FROM storage_connections)');
    }
    
    // Before deleting vendors: clean up transformation_mappings that reference vendor_agreements
    if (categories.includes('vendors')) {
      // Delete transformation_mappings that reference vendor_agreements
      await client.query('DELETE FROM transformation_mappings WHERE vendor_agreement_id IN (SELECT id FROM vendor_agreements)');
    }
    
    // Before deleting projects: clean up ALL references to projects being deleted (except Default Project)
    if (categories.includes('projects')) {
      // Delete files and all their children (regardless of files category selection)
      await client.query('DELETE FROM file_preview_data WHERE file_id IN (SELECT file_id FROM files WHERE project_id IN (SELECT id FROM projects WHERE id != 1))');
      await client.query('DELETE FROM file_columns WHERE file_id IN (SELECT file_id FROM files WHERE project_id IN (SELECT id FROM projects WHERE id != 1))');
      await client.query('DELETE FROM file_schemas WHERE file_id IN (SELECT file_id FROM files WHERE project_id IN (SELECT id FROM projects WHERE id != 1))');
      await client.query('DELETE FROM files WHERE project_id IN (SELECT id FROM projects WHERE id != 1)');
      
      // Delete transformations and children (regardless of transformations category selection)
      await client.query('DELETE FROM anomalies WHERE job_id IN (SELECT id::varchar FROM transformation_jobs WHERE project_id IN (SELECT id FROM projects WHERE id != 1))');
      await client.query('DELETE FROM profiling_metrics WHERE job_id IN (SELECT id::varchar FROM transformation_jobs WHERE project_id IN (SELECT id FROM projects WHERE id != 1))');
      await client.query('DELETE FROM transformation_logs WHERE job_id IN (SELECT id::varchar FROM transformation_jobs WHERE project_id IN (SELECT id FROM projects WHERE id != 1))');
      await client.query('DELETE FROM transformation_jobs WHERE project_id IN (SELECT id FROM projects WHERE id != 1)');
      await client.query('DELETE FROM transformation_mappings WHERE project_id IN (SELECT id FROM projects WHERE id != 1)');
      
      // Delete workflows and children (regardless of workflows category selection)
      await client.query('DELETE FROM workflow_executions WHERE project_id IN (SELECT id FROM projects WHERE id != 1)');
      await client.query('DELETE FROM workflow_steps WHERE project_id IN (SELECT id FROM projects WHERE id != 1)');
      await client.query('DELETE FROM n_step_workflow_runs WHERE workflow_id IN (SELECT id FROM n_step_workflows WHERE project_id IN (SELECT id FROM projects WHERE id != 1))');
      await client.query('DELETE FROM n_step_workflow_steps WHERE workflow_id IN (SELECT id FROM n_step_workflows WHERE project_id IN (SELECT id FROM projects WHERE id != 1))');
      await client.query('DELETE FROM n_step_workflows WHERE project_id IN (SELECT id FROM projects WHERE id != 1)');
      await client.query('DELETE FROM mapping_group_executions WHERE project_id IN (SELECT id FROM projects WHERE id != 1)');
      await client.query('DELETE FROM mapping_group_members WHERE group_id IN (SELECT id FROM mapping_groups WHERE project_id IN (SELECT id FROM projects WHERE id != 1))');
      await client.query('DELETE FROM mapping_groups WHERE project_id IN (SELECT id FROM projects WHERE id != 1)');
      await client.query('DELETE FROM schedules WHERE project_id IN (SELECT id FROM projects WHERE id != 1)');
      await client.query('DELETE FROM file_schedules WHERE project_id IN (SELECT id FROM projects WHERE id != 1)');
      await client.query('DELETE FROM job_schedules WHERE project_id IN (SELECT id FROM projects WHERE id != 1)');
      await client.query('DELETE FROM job_group_schedules WHERE project_id IN (SELECT id FROM projects WHERE id != 1)');
      await client.query('DELETE FROM bulk_profiling_jobs WHERE project_id IN (SELECT id FROM projects WHERE id != 1)');
      
      // Delete business processes
      await client.query('DELETE FROM business_processes WHERE project_id IN (SELECT id FROM projects WHERE id != 1)');
      
      // Delete logical file groups
      await client.query('DELETE FROM logical_file_groups WHERE project_id IN (SELECT id FROM projects WHERE id != 1)');
      
      // Delete connections that reference projects
      await client.query('DELETE FROM storage_connections WHERE project_id IN (SELECT id FROM projects WHERE id != 1)');
      await client.query('DELETE FROM transform_connections WHERE project_id IN (SELECT id FROM projects WHERE id != 1)');
      
      // Delete vendor agreements that reference projects
      await client.query('DELETE FROM vendor_agreement_documents WHERE agreement_id IN (SELECT id FROM vendor_agreements WHERE project_id IN (SELECT id FROM projects WHERE id != 1))');
      await client.query('DELETE FROM vendor_agreement_changes WHERE agreement_id IN (SELECT id FROM vendor_agreements WHERE project_id IN (SELECT id FROM projects WHERE id != 1))');
      await client.query('DELETE FROM vendor_agreement_approvals WHERE agreement_id IN (SELECT id FROM vendor_agreements WHERE project_id IN (SELECT id FROM projects WHERE id != 1))');
      await client.query('DELETE FROM vendor_agreements WHERE project_id IN (SELECT id FROM projects WHERE id != 1)');
      
      // Delete vendors that reference projects (and handle users that reference these vendors)
      await client.query('DELETE FROM vendor_file_profiling_results WHERE vendor_id IN (SELECT id FROM vendors WHERE project_id IN (SELECT id FROM projects WHERE id != 1))');
      await client.query('DELETE FROM vendor_users WHERE vendor_id IN (SELECT id FROM vendors WHERE project_id IN (SELECT id FROM projects WHERE id != 1))');
      // Set vendor_id to NULL for users that reference vendors being deleted (preserve users, remove vendor link)
      await client.query('UPDATE users SET vendor_id = NULL WHERE vendor_id IN (SELECT id FROM vendors WHERE project_id IN (SELECT id FROM projects WHERE id != 1))');
      await client.query('DELETE FROM vendors WHERE project_id IN (SELECT id FROM projects WHERE id != 1)');
      
      // Delete access control
      await client.query('DELETE FROM access_group_roles WHERE project_id IN (SELECT id FROM projects WHERE id != 1)');
      await client.query('DELETE FROM project_access_groups WHERE project_id IN (SELECT id FROM projects WHERE id != 1)');
      await client.query('DELETE FROM rbac_policies WHERE project_id IN (SELECT id FROM projects WHERE id != 1)');
    }
    
    // MAIN CATEGORY DELETIONS
    
    // Delete Files & Storage (cascades to related tables)
    if (categories.includes('files')) {
      const filePreviewResult = await client.query('DELETE FROM file_preview_data RETURNING preview_id');
      const fileColumnsResult = await client.query('DELETE FROM file_columns RETURNING id');
      const fileSchemasResult = await client.query('DELETE FROM file_schemas RETURNING id');
      const filesResult = await client.query('DELETE FROM files RETURNING id');
      
      deletedCounts.files = {
        files: filesResult.rowCount,
        file_columns: fileColumnsResult.rowCount,
        file_preview_data: filePreviewResult.rowCount,
        file_schemas: fileSchemasResult.rowCount
      };
    }
    
    // Delete Transformations & ETL
    if (categories.includes('transformations')) {
      const anomaliesResult = await client.query('DELETE FROM anomalies RETURNING id');
      const profilingResult = await client.query('DELETE FROM profiling_metrics RETURNING id');
      const logsResult = await client.query('DELETE FROM transformation_logs RETURNING id');
      const jobsResult = await client.query('DELETE FROM transformation_jobs RETURNING id');
      const transformationsResult = await client.query('DELETE FROM transformation_mappings RETURNING mapping_id');
      
      deletedCounts.transformations = {
        transformation_mappings: transformationsResult.rowCount,
        transformation_jobs: jobsResult.rowCount,
        transformation_logs: logsResult.rowCount,
        profiling_metrics: profilingResult.rowCount,
        anomalies: anomaliesResult.rowCount
      };
    }
