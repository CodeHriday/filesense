function generateCodeVerifier(length = 64) {
  const array = new Uint8Array(length);
  window.crypto.getRandomValues(array);
  return btoa(String.fromCharCode(...array))
    .replace(/\+/g, '-')
    .replace(/\//g, '_')
    .replace(/=+$/, '');
}
 
// Generate a code challenge from the verifier
async function generateCodeChallenge(verifier) {
  const encoder = new TextEncoder();
  const data = encoder.encode(verifier);
  const digest = await window.crypto.subtle.digest('SHA-256', data);
  const base64Digest = btoa(String.fromCharCode(...new Uint8Array(digest)))
    .replace(/\+/g, '-')
    .replace(/\//g, '_')
    .replace(/=+$/, '');
  return base64Digest;
}
 
/**
 * Initiate Okta login
 */
async function initiateOktaLogin(){
  const verifier = generateCodeVerifier();
    const challenge = await generateCodeChallenge(verifier);
    
    document.cookie = `codeVerifier=${verifier}; path=/; secure; samesite=strict`;
 
    localStorage.setItem('pkce_verifier', verifier);
    const oktaDomain = getSecret('OKTA_DOMAIN');
    const clientId = getSecret('OKTA_CLIENT_ID');
    
    const authUrl = `${oktaDomain}/oauth2/default/v1/authorize?` +
      new URLSearchParams({
        client_id: clientId || '',
        response_type: 'code',
        scope: 'openid profile email',
        redirect_uri: getSecret('OKTA_REDIRECT_URI') || 'http://localhost:5000/auth/callback',
        state: 'xyz',
        code_challenge: challenge,
        code_challenge_method: 'S256'
      });
    window.location.href = authUrl;
};
 
// Protected route component
const ProtectedRoute = ({ children }) => {
  const { currentUser, authStatus } = useAuth();
  const location = useLocation();
  const pathname = location.pathname;
  
  // Use authStatus for atomic routing decisions
  // This prevents the race condition where loading=false before currentUser is set
  if (authStatus === AUTH_STATUS.CHECKING) {
    return <div className="loading-screen">Loading...</div>;
  }
  
  if (authStatus === AUTH_STATUS.UNAUTHENTICATED || !currentUser) {
    // Pass current location so Login can redirect back after authentication
    return <Navigate to="/login" state={{ from: location }} replace />;
  }
  
  // Extract route parameters to pass to children if it's a function
  // This handles routes like /profile/:fileId
  if (typeof children === 'function') {
    // Extract fileId from the path based on various possible route patterns
    let fileId = null;
    const profileMatch = pathname.match(/\/profile\/([^\/]+)/);
    const qualityMatch = pathname.match(/\/quality\/([^\/]+)/);
    const reportMatch = pathname.match(/\/report\/([^\/]+)/);
    const lineageMatch = pathname.match(/\/lineage\/impact\/([^\/]+)/);
    
    if (profileMatch) fileId = profileMatch[1];
    else if (qualityMatch) fileId = qualityMatch[1];
    else if (reportMatch) fileId = reportMatch[1];
    else if (lineageMatch) fileId = lineageMatch[1];
    
    // Also validate fileId format is reasonable (not "undefined" as string)
    if (fileId === 'undefined') fileId = null;
    
    return children({ fileId });
  }
  
  return children;
};
ProtectedRoute.propTypes = {
  children: PropTypes.oneOfType([
    PropTypes.node,
    PropTypes.func
  ]).isRequired
};

                <Route path="/api-documentation" element={<ProtectedRoute><APIDocumentation /></ProtectedRoute>} />
                <Route path="/projects" element={<ProtectedRoute><Projects /></ProtectedRoute>} />
                <Route path="/project-management" element={<ProtectedRoute><ProjectManagement /></ProtectedRoute>} />
                <Route path="/group-management" element={<ProtectedRoute><GroupManagement /></ProtectedRoute>} />
                <Route path="/system-cleanup" element={<ProtectedRoute><SystemCleanup /></ProtectedRoute>} />
                <Route path="/alerts-config" element={<ProtectedRoute><AlertsConfig /></ProtectedRoute>} />
                <Route path="/alert-history" element={<ProtectedRoute><AlertHistory /></ProtectedRoute>} />
                <Route path="/email-providers" element={<ProtectedRoute><EmailProvidersConfig /></ProtectedRoute>} />
                <Route path="/oauth2-callback" element={<OAuth2Callback />} />
                {/* ARCHIVED: <Route path="/workflow-canvas" element={<ProtectedRoute><WorkflowCanvasStandalone /></ProtectedRoute>} /> */}
                {/* ARCHIVED: <Route path="/workflow-test" element={<ProtectedRoute><WorkflowTest /></ProtectedRoute>} /> */}
                <Route path="/documentation" element={<ProtectedRoute><Documentation /></ProtectedRoute>} />
                <Route path="/help" element={<ProtectedRoute><Documentation /></ProtectedRoute>} />
                {/* Development Routes - ARCHIVED */}

                <Route path="/performance-optimizations" element={<ProtectedRoute><PerformanceOptimizations /></ProtectedRoute>} />
                <Route path="/login" element={currentUser ? <Navigate to="/" replace /> : <Login />} />
                <Route path="*" element={<Navigate to="/" replace />} />
              </Routes>
              </Suspense>
            </main>
          </div>
          
          {/* AI Insights Sidebar - only show when toggled */}
          {showAIInsights && selectedFileId && (
            <div className="ai-insights-container">
              {/* Import dynamically to reduce initial load time */}
              {React.lazy(() => import('./components/AIPredictiveInsights')).render({
                fileId: selectedFileId,
                onClose: () => setShowAIInsights(false)
              })}
            </div>
          )}
        </>
      )}
      
      {/* Login route when not logged in - only show after auth check confirms unauthenticated */}
      {authStatus === AUTH_STATUS.UNAUTHENTICATED && (
        <Routes>
          <Route path="/login" element={<Login />} />
          <Route path="/auth/callback" element={<OktaCallback />} />
          <Route path="*" element={<Navigate to="/login" state={{ from: location }} replace />} />
        </Routes>
      )}
      
      {/* Accessibility Widget - hidden */}
      
      {/* Onboarding Wizard removed - no longer displayed */}
      
      {/* Global Notifications */}
      <NotificationList />
      
      {/* Toast Notifications */}
      <ToastContainer />
    </div>
  );
}
